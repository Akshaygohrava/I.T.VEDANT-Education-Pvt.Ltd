📐 design & relationships
Normalization · organize to reduce redundancy
  • 🔹 1:1 — person ↔ pan card
  • 🔹 1:N — school ↔ student
  • 🔹 N:N — customers ↔ products
🔑 PRIMARY = NOT NULL + UNIQUE (one per table)
🔗 FOREIGN KEY : link tables, many FKs allowed, references PK
CREATE TABLE department (
  Id INT PRIMARY KEY,
  Name VARCHAR(100) NOT NULL
);
CREATE TABLE employee (
  Id INT PRIMARY KEY,
  Name VARCHAR(100) NOT NULL,
  Salary INT, Department_id INT,
  CONSTRAINT emp_depart -- FK name
  FOREIGN KEY (Department_id) REFERENCES department(Id)
);
DESC employee;

✏️ ALTER table
-- ADD column
ALTER TABLE student ADD COLUMN address VARCHAR(250) NOT NULL;
-- DROP column
ALTER TABLE student DROP COLUMN address;
-- ADD AFTER another column
ALTER TABLE student ADD COLUMN Marks INT AFTER Email;
-- ADD FIRST
ALTER TABLE student ADD COLUMN Course_Id INT FIRST;
-- RENAME column
ALTER TABLE student RENAME COLUMN DOB TO Date_of_Birth;
-- MODIFY datatype
ALTER TABLE student MODIFY COLUMN Age VARCHAR(100);
-- ADD CONSTRAINT (e.g. UNIQUE)
ALTER TABLE student ADD CONSTRAINT -- (unique constraint)
⚠️ CONSTRAINTS
CREATE TABLE check_default (
  col_1 INT PRIMARY KEY AUTO_INCREMENT,
  col_2 VARCHAR(100) NOT NULL,
  col_3 CHAR(10) UNIQUE,
  col_4 INT CHECK (col_1 > 3),
  col_5 INT DEFAULT 111
);
CREATE TABLE check_constraints (
  Id INT PRIMARY KEY AUTO_INCREMENT,
  Name VARCHAR(100) NOT NULL,
  Phone CHAR(10) UNIQUE,
  Email VARCHAR(50) UNIQUE NOT NULL,
  Age INT CHECK (Age > 18),
  Date DATE DEFAULT (CURRENT_DATE)
);
✅ CHECK🔁 DEFAULT🔄 AUTO_INCREMENT
⚡ INDEX & VIEW
-- CREATE INDEX
CREATE INDEX demo ON student (Id, Name, Age, Email);
-- DROP INDEX

-- CREATE VIEW
CREATE VIEW demo_view AS SELECT Id, Name, Email, Age FROM student;

  CREATE TABLE demo_db.demo_table (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    Email VARCHAR(100) UNIQUE NOT NULL
  );

  VALUES

  (2, "Arjun Gohrava", "arjun@gmail.com");

DELIMITER ;
CALL demo_procedure;
