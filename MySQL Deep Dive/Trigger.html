🔄 automatic 📌 INSERT/UPDATE/DELETE ⏰ BEFORE / AFTER 📦 audit log 🛡️ data integrity

📐 Syntax & Timing
CREATE TRIGGER trigger_name
{ BEFORE | AFTER } { INSERT | UPDATE | DELETE }
ON table_name FOR EACH ROW
BEGIN
-- SQL statements
-- OLD.column (for UPDATE/DELETE)
-- NEW.column (for INSERT/UPDATE)
END;
⏰ BEFORE: validate/modify data before write · AFTER: log/audit after write
🔹 INSERT 🔹 UPDATE 🔹 DELETE 🔹 FOR EACH ROW
🛡️ BEFORE INSERT · validation
-- ensure age >= 18
DELIMITER //
CREATE TRIGGER check_age_before_insert
BEFORE INSERT ON users
FOR EACH ROW
BEGIN
IF NEW.age < 18 THEN
SIGNAL SQLSTATE '45000'
SET MESSAGE_TEXT = 'Age must be >= 18';
END IF;
END //
DELIMITER ;
-- auto-format email to lowercase
CREATE TRIGGER format_email
BEFORE INSERT ON users
FOR EACH ROW
BEGIN
SET NEW.email = LOWER(NEW.email);
END;
✅ Prevents invalid data · modifies NEW values before insert
📋 AFTER INSERT · audit log
-- create audit table
CREATE TABLE user_audit (
log_id INT AUTO_INCREMENT PRIMARY KEY,
user_id INT, action VARCHAR(20),
changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
-- trigger: log every new user
DELIMITER //
CREATE TRIGGER log_user_insert
AFTER INSERT ON users
FOR EACH ROW
BEGIN
INSERT INTO user_audit (user_id, action)
VALUES (NEW.id, 'INSERT');
END //
DELIMITER ;
📌 Great for tracking changes, compliance, and history
✏️ BEFORE UPDATE · modify & validate
-- auto-update last_modified timestamp
DELIMITER //
CREATE TRIGGER update_timestamp
BEFORE UPDATE ON users
FOR EACH ROW
BEGIN
SET NEW.last_modified = NOW();
END //
DELIMITER ;
-- prevent lowering salary by more than 10%
CREATE TRIGGER salary_check
BEFORE UPDATE ON employees
FOR EACH ROW
BEGIN
IF NEW.salary < OLD.salary * 0.9 THEN
SIGNAL SQLSTATE '45000'
SET MESSAGE_TEXT = 'Salary cut exceeds 10%';
END IF;
END;
🔍 OLD.column = value before update · NEW.column = new value
🗑️ AFTER DELETE · cleanup
-- archive deleted records
CREATE TABLE deleted_users_archive (
id INT, name VARCHAR(100), email VARCHAR(100),
deleted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
-- trigger: archive before delete
DELIMITER //
CREATE TRIGGER archive_user
AFTER DELETE ON users
FOR EACH ROW
BEGIN
INSERT INTO deleted_users_archive (id, name, email)
VALUES (OLD.id, OLD.name, OLD.email);
END //
DELIMITER ;
💡 OLD contains the row that was deleted · useful for recovery
⚖️ OLD vs NEW · DROP
OperationOLD available?NEW available?
INSERT❌✅
UPDATE✅ (old values)✅ (new values)
DELETE✅❌
-- Drop a trigger
DROP TRIGGER [IF EXISTS] trigger_name;
-- Show all triggers
SHOW TRIGGERS; -- or
SELECT * FROM information_schema.triggers
WHERE trigger_schema = 'your_database';
📌 Triggers run automatically · no explicit call · can be disabled with DISABLE TRIGGER
✅ GOOD: audit logs, validation, auto-updates, data integrity
❌ AVOID: complex business logic, recursive triggers, heavy processing
-- Multiple statements in a trigger
DELIMITER //
CREATE TRIGGER multi_action
AFTER INSERT ON orders
FOR EACH ROW
BEGIN
UPDATE inventory SET quantity = quantity - NEW.qty WHERE product_id = NEW.product_id;
INSERT INTO order_log (order_id, status) VALUES (NEW.id, 'placed');
END //
DELIMITER ;
-- BEFORE DELETE: prevent deletion of active users
CREATE TRIGGER prevent_delete_active
BEFORE DELETE ON users
FOR EACH ROW
BEGIN
IF OLD.status = 'active' THEN
SIGNAL SQLSTATE '45000'
SET MESSAGE_TEXT = 'Cannot delete active user';
END IF;
END;
