🎯 Basic Syntax
DELIMITER //

RETURNS return_datatype

BEGIN
-- SQL statements

📌 Think of it like a calculator: inputs → one result
-- 1. Add Two Numbers (simplest)
DELIMITER //
CREATE FUNCTION add_numbers(a INT, b INT)
RETURNS INT DETERMINISTIC
BEGIN
RETURN a + b;
END //
DELIMITER ;
SELECT add_numbers(5, 3); -- 8
SELECT add_numbers(10, 20); -- 30
👋 Greeting & Age
-- 2. Greeting Function
DELIMITER //
CREATE FUNCTION greet(name VARCHAR(50))
RETURNS VARCHAR(100) DETERMINISTIC
BEGIN
RETURN CONCAT('Hello, ', name, '! Welcome!');
END //
DELIMITER ;
SELECT greet('John'); -- "Hello, John! Welcome!"
SELECT greet('Alice'); -- "Hello, Alice! Welcome!"
-- 3. Calculate Age
DELIMITER //
CREATE FUNCTION calculate_age(birth_date DATE)
RETURNS INT DETERMINISTIC
BEGIN
RETURN YEAR(CURDATE()) - YEAR(birth_date);
END //
DELIMITER ;
SELECT calculate_age('1990-05-15'); -- 36 (2026)
💰 Tax & Discount
-- 4. Calculate Tax
DELIMITER //
CREATE FUNCTION calculate_tax(amount DECIMAL(10,2), tax_rate DECIMAL(5,2))
RETURNS DECIMAL(10,2) DETERMINISTIC
BEGIN
RETURN amount * (tax_rate / 100);
END //
DELIMITER ;
SELECT calculate_tax(1000, 10); -- 100.00
SELECT calculate_tax(500.50, 18); -- 90.09
-- 5. Get Discount Amount (final price)
DELIMITER //
CREATE FUNCTION get_discount(price DECIMAL(10,2), discount_percent INT)
RETURNS DECIMAL(10,2) DETERMINISTIC
BEGIN
DECLARE discount_amount DECIMAL(10,2);
SET discount_amount = price * (discount_percent / 100);
RETURN price - discount_amount;
END //
DELIMITER ;
SELECT get_discount(100, 20); -- 80.00
SELECT get_discount(250, 15); -- 212.50
🔢 Even/Odd & Full Name
-- 6. Check if Number is Even
DELIMITER //
CREATE FUNCTION is_even(num INT)
RETURNS VARCHAR(10) DETERMINISTIC
BEGIN
IF num % 2 = 0 THEN RETURN 'Even';
ELSE RETURN 'Odd';
END IF;
END //
DELIMITER ;
SELECT is_even(4); -- "Even"
SELECT is_even(7); -- "Odd"
-- 7. Get Full Name from ID (uses table)
CREATE TABLE users (
id INT, first_name VARCHAR(50), last_name VARCHAR(50)
);
INSERT INTO users VALUES (1, 'John', 'Doe'), (2, 'Jane', 'Smith');
DELIMITER //
CREATE FUNCTION get_full_name(user_id INT)
RETURNS VARCHAR(100) DETERMINISTIC
BEGIN
DECLARE full_name VARCHAR(100);
SELECT CONCAT(first_name, ' ', last_name) INTO full_name
FROM users WHERE id = user_id;
RETURN full_name;
END //
DELIMITER ;
SELECT get_full_name(1); -- "John Doe"
SELECT get_full_name(2); -- "Jane Smith"

-- 9. Calculate Bonus with Conditions
DELIMITER //
CREATE FUNCTION calculate_bonus(salary DECIMAL(10,2), performance_score INT)
RETURNS DECIMAL(10,2) DETERMINISTIC
BEGIN
DECLARE bonus DECIMAL(10,2);
IF performance_score >= 90 THEN SET bonus = salary * 0.20;
ELSEIF performance_score >= 70 THEN SET bonus = salary * 0.10;
ELSE SET bonus = 0;
END IF;
RETURN bonus;
END //
DELIMITER ;
SELECT calculate_bonus(50000, 95); -- 10000.00
SELECT calculate_bonus(50000, 75); -- 5000.00
SELECT calculate_bonus(50000, 50); -- 0.00
📊 Grade & Initials
-- 10. Grade Calculator
DELIMITER //
CREATE FUNCTION get_grade(score INT)
RETURNS VARCHAR(2) DETERMINISTIC
BEGIN
DECLARE grade VARCHAR(2);
CASE
WHEN score >= 90 THEN SET grade = 'A';
WHEN score >= 80 THEN SET grade = 'B';
WHEN score >= 70 THEN SET grade = 'C';
WHEN score >= 60 THEN SET grade = 'D';
ELSE SET grade = 'F';
END CASE;
RETURN grade;
END //
DELIMITER ;
SELECT get_grade(95); -- "A"
SELECT get_grade(72); -- "C"
-- 11. Extract Initials
DELIMITER //
CREATE FUNCTION extract_initials(full_name VARCHAR(100))
RETURNS VARCHAR(20) DETERMINISTIC
BEGIN
DECLARE first_initial CHAR(1);
DECLARE last_initial CHAR(1);
SET first_initial = UPPER(SUBSTRING(full_name, 1, 1));
SET last_initial = UPPER(SUBSTRING(full_name, LOCATE(' ', full_name) + 1, 1));
RETURN CONCAT(first_initial, last_initial);
END //
DELIMITER ;
SELECT extract_initials('John Doe'); -- "JD"
SELECT extract_initials('Jane Smith'); -- "JS"

-- ❌ BAD: function cannot modify data
CREATE FUNCTION bad_function(user_id INT)
RETURNS VARCHAR(100)
BEGIN
UPDATE users SET last_login = NOW() WHERE id = user_id; -- ERROR!
RETURN 'Updated';
END //
-- ✅ GOOD: only read data
CREATE FUNCTION good_function(user_id INT)
RETURNS VARCHAR(100)
BEGIN
DECLARE user_name VARCHAR(100);
SELECT name INTO user_name FROM users WHERE id = user_id;
RETURN user_name;
END //
🔥 E‑commerce System
-- Setup tables
CREATE TABLE products (
id INT PRIMARY KEY, name VARCHAR(100), price DECIMAL(10,2), category VARCHAR(50)
);
CREATE TABLE orders (
id INT PRIMARY KEY, product_id INT, quantity INT, order_date DATE
);
-- 1. Final price with tax
DELIMITER //
CREATE FUNCTION final_price(price DECIMAL(10,2), tax_rate DECIMAL(5,2))
RETURNS DECIMAL(10,2) DETERMINISTIC
BEGIN RETURN price + (price * tax_rate / 100); END //
DELIMITER ;
-- 2. Order total
DELIMITER //
CREATE FUNCTION order_total(order_id INT)
RETURNS DECIMAL(10,2) DETERMINISTIC
BEGIN
DECLARE total DECIMAL(10,2);
SELECT SUM(p.price * o.quantity) INTO total
FROM orders o JOIN products p ON o.product_id = p.id
WHERE o.id = order_id;
RETURN IFNULL(total, 0);
END //
DELIMITER ;
-- 3. Discount based on quantity
DELIMITER //
CREATE FUNCTION apply_discount(quantity INT, price DECIMAL(10,2))
RETURNS DECIMAL(10,2) DETERMINISTIC
BEGIN
IF quantity >= 10 THEN RETURN price * 0.85;
ELSEIF quantity >= 5 THEN RETURN price * 0.90;
ELSE RETURN price;
END IF;
END //
DELIMITER ;
📌 Use all together: SELECT p.name, apply_discount(10, p.price) AS discounted, final_price(..., 12) AS final_with_tax, order_total(1) FROM products p;
🚀 Quick Reference Card
-- Basic function
CREATE FUNCTION func_name(param1 INT) RETURNS INT DETERMINISTIC
BEGIN RETURN param1 * 2; END;
-- With variable
CREATE FUNCTION func_name(param1 INT) RETURNS INT DETERMINISTIC
BEGIN DECLARE result INT; SET result = param1 * 2; RETURN result; END;
-- With conditional
CREATE FUNCTION func_name(score INT) RETURNS VARCHAR(10) DETERMINISTIC
BEGIN IF score > 80 THEN RETURN 'Pass'; ELSE RETURN 'Fail'; END IF; END;
-- With table data
CREATE FUNCTION func_name(id INT) RETURNS VARCHAR(50) DETERMINISTIC
BEGIN DECLARE val VARCHAR(50); SELECT name INTO val FROM table WHERE id = id; RETURN val; END;
💡 Key Takeaway: Function = Calculator that always returns one value. Reusable, used in SELECT, no data modification.
📌 WHEN TO USE: calculations, formatting, validation, reusable logic in queries
❌ AVOID: data modification, complex transactions, sending emails/APIs

-- Demonstrate DETERMINISTIC vs NOT
-- DETERMINISTIC: same input → same output (e.g. add_numbers)
-- NOT DETERMINISTIC: can return different values (e.g. using RAND(), NOW())
CREATE FUNCTION random_bonus(salary DECIMAL(10,2))
RETURNS DECIMAL(10,2) NOT DETERMINISTIC
BEGIN RETURN salary * RAND(); END;
